    29→function resolveElectronBuilderWindowsTarget({
    30→  electronPlatformName,
    31→  arch,
    32→  configuredTargetPlatform,
    33→}) {
    34→  if (electronPlatformName !== "win32") {
    35→    throw new Error(
    36→      `[electron-builder.config] context platform is not win32: ${String(electronPlatformName)}`,
    37→    );
    38→  }
    39→  const actualArch = ELECTRON_BUILDER_ARCH[arch];
    40→  if (!actualArch) {
    41→    throw new Error(
    42→      `[electron-builder.config] unsupported electron-builder Windows architecture: ${String(arch)}`,
    43→    );
    44→  }
    45→  const actualTarget = {
    46→    os: "win32",
    47→    arch: actualArch,
    48→    key: `win32-${actualArch}`,
    49→  };
    50→  if (
    51→    configuredTargetPlatform?.os !== actualTarget.os ||
    52→    configuredTargetPlatform?.arch !== actualTarget.arch ||
    53→    configuredTargetPlatform?.key !== actualTarget.key
    54→  ) {
    55→    throw new Error(
    56→      `[electron-builder.config] configured target ${String(configuredTargetPlatform?.key)} does not match electron-builder target ${actualTarget.key}`,
    57→    );
    58→  }
    59→  return actualTarget;
    60→}

   173→async function writeWindowsInstallManifest(context) {
   174→  if (context.electronPlatformName !== "win32") return;
   175→
   176→  const root = context.appOutDir;
   177→  const files = [];
   178→  const visit = async (directory, relativeDirectory = "") => {
   179→    for (const entry of await readdir(directory, { withFileTypes: true })) {
   180→      const relativePath = relativeDirectory ? `${relativeDirectory}/${entry.name}` : entry.name;
   181→      const absolutePath = join(directory, entry.name);
   182→      if (entry.isDirectory()) {
   183→        await visit(absolutePath, relativePath);
   184→      } else if (entry.isFile() || entry.isSymbolicLink()) {
   185→        files.push(relativePath.replaceAll("/", "\\"));
   186→      }
   187→    }
   188→  };
   189→
   190→  await visit(root);
   191→  files.sort();
   192→  await writeFile(join(root, WINDOWS_INSTALL_MANIFEST_NAME), `${files.join("\r\n")}\r\n`, "utf8");
   193→}

   314→function resolveMissingRuntimeModules(appAsarPath) {
   315→  const asarEntries = runAsarCommandAndReadStdout(["list", appAsarPath])
   316→    .split("\n")
   317→    .map(normalizeAsarEntry)
   318→    .filter(Boolean);
   319→  const asarEntrySet = new Set(asarEntries);
   320→
   321→  const runtimeModules = collectRuntimeModuleClosureEntries(
   322→    REQUIRED_ASAR_RUNTIME_MODULES,
   323→    runtimeModuleLookupRoots,
   324→  );
   325→  const resolvableRuntimeModules = runtimeModules.filter((entry) => {
   326→    if (!entry.sourceModulePath) {
   327→      // 不同平台/安装布局下，部分运行时依赖可能被裁剪或未落到本次打包工作区。
   328→      // 之前这里直接在 copy 阶段抛错会中断整个平台出包；改为记录告警并跳过该模块，
   329→      // 让 afterPack 只处理当前环境确实可解析的依赖，避免 CI 因单个可选依赖缺失全量失败。
   330→      console.warn(
   331→        `[afterPack] runtime module not found, skip injection: ${entry.moduleName}; searched=${runtimeModuleLookupRoots
   332→          .map((lookupRoot) => resolve(lookupRoot, "node_modules", entry.moduleName))
   333→          .join(", ")}`,
   334→      );
   335→      return false;
   336→    }
   337→    return true;
   338→  });
   339→  return resolvableRuntimeModules.filter((entry) => {
   340→    const { moduleName } = entry;
   341→    const moduleRoot = `/node_modules/${moduleName}`;
   342→    if (asarEntrySet.has(moduleRoot)) {
   343→      return false;
   344→    }
   345→    for (const entry of asarEntrySet) {
   346→      if (entry.startsWith(`${moduleRoot}/`)) {
   347→        return false;
   348→      }
   349→    }
   350→    return true;
   351→  });
   352→}

   354→async function injectHoistedRuntimeModulesIntoAsar(context) {
   355→  const appAsarPath = resolveAppAsarPath(context);
   356→  if (!existsSync(appAsarPath)) {
   357→    throw new Error(`打包产物缺少 app.asar: ${appAsarPath}`);
   358→  }
   359→
   360→  const missingRuntimeModules = runTimedSync("afterPack:scan-missing-runtime-modules", () =>
   361→    resolveMissingRuntimeModules(appAsarPath),
   362→  );
   363→  if (missingRuntimeModules.length === 0) {
   364→    // 之前 afterPack 每次都完整 extract/pack app.asar，即使运行时依赖已经齐全也会重复重写。
   365→    // 这会把每次打包固定拉长十几秒到几十秒。先做缺失扫描，只有真的缺包才执行重写流程。
   366→    console.log("[afterPack] runtime modules already complete, skip app.asar rewrite");
   367→    return;
   368→  }
   369→  console.log(`[afterPack] missing runtime modules count=${missingRuntimeModules.length}`);
   370→
   371→  // CI 会把 TMPDIR 指到项目内 .tmp，GitLab get_sources/clean 可能在脚本启动前清掉该目录。
   372→  // afterPack 里重写 app.asar 同样依赖 mkdtempSync，必须自己兜底创建父目录，避免后续签名阶段只看到 .app 消失。
   373→  mkdirSync(tmpdir(), { recursive: true });
   374→  const stagingDir = mkdtempSync(resolve(tmpdir(), "zcode-app-asar-"));
   375→  try {
   376→    runTimedSync("afterPack:asar-extract", () =>
   377→      runAsarCommand(["extract", appAsarPath, stagingDir]),
   378→    );
   379→
   380→    const stagingNodeModulesDir = resolve(stagingDir, "node_modules");
   381→    mkdirSync(stagingNodeModulesDir, { recursive: true });
   382→
   383→    runTimedSync("afterPack:copy-runtime-modules", () => {
   384→      for (const runtimeModule of missingRuntimeModules) {
   385→        const { moduleName, sourceModulePath } = runtimeModule;
   386→        const targetModulePath = resolve(stagingNodeModulesDir, moduleName);
   387→
   388→        if (!sourceModulePath) {
   389→          throw new Error(
   390→            `未找到运行时依赖 ${moduleName}，已搜索: ${runtimeModuleLookupRoots
   391→              .map((lookupRoot) => resolve(lookupRoot, "node_modules", moduleName))
   392→              .join(", ")}`,
   393→          );
   394→        }
   395→
   396→        // pnpm hoisted 布局下，electron-builder 可能把主包打进 app.asar，
   397→        // 却漏掉它解析时还要去根 node_modules 找的运行时依赖。
   398→        // 之前 require-in-the-middle 漏过 module-details-from-path，这次 @fiahfy/icns 又漏了 pngjs，
   399→        // 最终都会在已安装应用里触发 Cannot find module 并让主进程启动直接崩溃。
   400→        // 这里按 package.json 递归补齐依赖闭包，避免每次只补一个缺失包、上线后再暴露下一个子依赖。
   401→        // 只靠 package.json 显式依赖、本包 node_modules 镜像、files include 都没让它稳定进 asar，
   402→        // 所以在 afterPack 阶段直接重写 app.asar，先把这些运行时包补进去，再交给后续签名和出包。
